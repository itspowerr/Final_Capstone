\section{Module M1 --- Wallet Authentication}

% ──────────────────────────────────────────────────────────────────────────
\subsection{TC ID: TC-M1-01 --- MetaMask Auth: Nonce Challenge and Signature Verification}

\begin{tabular}{|p{0.22\textwidth}|p{0.74\textwidth}|}
\hline
\rowcolor{lightblue} \textbf{Field} & \textbf{Value} \\
\hline
Req Mapping & FR-AUTH-01, FR-SEC-02 \\
\hline
Priority & \sevhigh \\
\hline
Technique & Functional --- Valid partition \\
\hline
Precondition & MetaMask installed and unlocked; backend running on port 8000; Redis available for nonce storage; client wallet seeded in PostgreSQL (wallet\_address = \texttt{0x70997970c51812dc3a010c7d01b50e0d17dc79c8}) \\
\hline
Test Steps &
\begin{enumerate}[nosep, leftmargin=*]
  \item Frontend calls \texttt{GET /api/auth/wallet/challenge?address=0x7099...c8}
  \item Backend generates 64-character hex nonce
  \item Nonce stored in Redis with key \texttt{nonce:0x7099...c8}
  \item Frontend calls \texttt{POST /api/auth/wallet/login} with address, signature, and role
  \item Backend recovers public key from signature using \texttt{eth\_account.Account.recover\_message}
  \item Backend compares recovered address to claimed address
\end{enumerate} \\
\hline
Test Data &
\begin{itemize}[nosep, leftmargin=*]
  \item Valid Ethereum address (checksummed): \texttt{0x70997970c51812dc3a010c7d01b50e0d17dc79c8}
  \item Valid EIP-191 signature (65 bytes, 130 hex chars)
  \item Role: \texttt{``client''}
\end{itemize} \\
\hline
Expected Result &
\begin{itemize}[nosep, leftmargin=*]
  \item HTTP 200 with JSON body
  \item Response contains \texttt{access\_token} and \texttt{token\_type: ``bearer''}
  \item Response contains \texttt{user.id} matching seeded client
  \item Nonce deleted from Redis after successful verification
\end{itemize} \\
\hline
Pass/Fail Criteria &
\begin{itemize}[nosep, leftmargin=*]
  \item \passresult{} if HTTP 200 + token returned + nonce consumed from Redis
  \item \failresult{} if HTTP 401/403 or nonce not deleted
\end{itemize} \\
\hline
\end{tabular}
